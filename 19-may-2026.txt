    FIRST_NAME CHAR(25),
    LAST_NAME CHAR(25),
    SALARY INT(15),
    JOINING_DATE DATETIME,
    DEPARTMENT CHAR(25)

SELECT *
FROM Worker
WHERE DEPARTMENT = 'Admin'
AND SALARY > 100000;

SELECT *
FROM Worker
WHERE WORKER_ID IN (1, 4, 6);

SELECT *
FROM Worker
WHERE WORKER_ID NOT IN (2, 5, 8);

SELECT *
FROM Worker
WHERE SALARY BETWEEN 80000 AND 300000;

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '_i%';

SELECT *
FROM Worker
WHERE LAST_NAME LIKE '%h%';

SELECT *
FROM Worker
WHERE DEPARTMENT LIKE 'A%';
SELECT MAX(SALARY) AS MAX_SALARY
FROM Worker;

SELECT MAX(WORKER_ID) AS MAX_WORKER_ID
FROM Worker;

SELECT MAX(JOINING_DATE) AS LATEST_JOINING
FROM Worker;

SELECT MIN(SALARY) AS MIN_SALARY
FROM Worker;

SELECT MIN(WORKER_ID) AS MIN_WORKER_ID
FROM Worker;

SELECT MIN(JOINING_DATE) AS EARLIEST_JOINING
FROM Worker;

SELECT SUM(SALARY) AS TOTAL_SALARY
FROM Worker;

SELECT SUM(WORKER_ID) AS TOTAL_WORKER_ID
FROM Worker;

SELECT SUM(SALARY) AS ADMIN_TOTAL_SALARY
FROM Worker
WHERE DEPARTMENT='Admin';

SELECT AVG(SALARY) AS AVG_SALARY
FROM Worker;

SELECT AVG(WORKER_ID) AS AVG_WORKER_ID
FROM Worker;

SELECT AVG(SALARY) AS HR_AVG_SALARY
FROM Worker
WHERE DEPARTMENT='HR';

SELECT COUNT(*) AS TOTAL_WORKERS
FROM Worker;

SELECT COUNT(DEPARTMENT) AS TOTAL_DEPARTMENTS
FROM Worker;

SELECT COUNT(*) AS ADMIN_WORKERS
FROM Worker
WHERE DEPARTMENT='Admin';
SELECT *
FROM Worker
WHERE DEPARTMENT = 'Admin'

FROM Worker

SELECT WORKER_ID AS ROLL_NUMBER
FROM Worker;
SELECT DISTINCT DEPARTMENT
FROM Worker;
SELECT FIRST_NAME, SALARY,
CASE
    WHEN SALARY >= 300000 THEN 'High Salary'
    WHEN SALARY >= 100000 THEN 'Medium Salary'
    ELSE 'Low Salary'
END AS SALARY_STATUS
FROM Worker;
CREATE VIEW Admin_Workers AS
SELECT *
FROM Worker
WHERE DEPARTMENT = 'Admin';

CREATE VIEW HR_Workers AS
SELECT *
FROM Worker
WHERE DEPARTMENT = 'HR';

CREATE VIEW High_Salary AS
SELECT *
FROM Worker
WHERE SALARY > 200000;

CREATE VIEW Worker_Names AS
SELECT FIRST_NAME, LAST_NAME
FROM Worker;
