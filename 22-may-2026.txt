create table departments(
  did int primary key,
  dname varchar(255) not null unique
);

select * from departments order by did asc;

create table HOD
(Hid int primary key,
hname varchar(20) not null unique
);

insert into HOD values (101,'Arun'),(102,'Jayant'),(104,'Karthik'),(105,'Murali');

select * from HOD;

SELECT * FROM 
departments CROSS JOIN HOD order by did asc;

SELECT * FROM 
departments INNER JOIN HOD where departments.did=HOD.hid;


SELECT * FROM 
departments natural JOIN HOD;

SELECT * FROM 
departments LEFT OUTER JOIN HOD ON (departments.did=HOD.hid) order by did asc;

SELECT * FROM 
departments FULL JOIN HOD ON (departments.did=HOD.hid);
